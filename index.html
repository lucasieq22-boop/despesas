async function handleSubmit(event) { event.preventDefault(); const linhaInput = document.getElementById('linha'); const btn = document.getElementById('btn-salvar'); const msg = document.getElementById('mensagem'); const texto = linhaInput.value.trim(); if (!texto) { msg.textContent = 'Informe valor e descrição na mesma linha.'; msg.className = 'msg erro'; return; } btn.disabled = true; btn.textContent = 'Salvando...'; msg.textContent = ''; msg.className = 'msg'; try { const response = await fetch(API_URL, { method: 'POST', redirect: 'follow', // Segue o redirecionamento 302 obrigatório do Apps Script headers: { 'Content-Type': 'text/plain;charset=utf-8' // Crucial: evita bloqueio de pré-flight do CORS }, body: JSON.stringify({ texto: texto }) }); const textoResposta = await response.text(); let retorno; try { retorno = JSON.parse(textoResposta); } catch (e) { // Se não conseguir converter para JSON mas o status for 200/OK, considera salvo retorno = { ok: true, valor: '', categoria: 'Lançado' }; } if (retorno.ok) { let detalhes = ''; if (retorno.valor) { detalhes += `Valor: R$ ${Number(retorno.valor).toFixed(2).replace('.', ',')} | `; } if (retorno.categoria) { detalhes += `Categoria: ${retorno.categoria}`; } msg.textContent = `Despesa salva! ${detalhes}`.trim(); msg.className = 'msg ok'; linhaInput.value = ''; linhaInput.focus(); } else { throw new Error(retorno.message || 'Erro ao processar dados.'); } } catch (erro) { msg.textContent = erro.message || 'Erro de comunicação com a planilha.'; msg.className = 'msg erro'; } finally { btn.disabled = false; btn.textContent = 'Salvar despesa'; } }
